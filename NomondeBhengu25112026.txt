USE master;

WITH PercentageMarks AS (
    SELECT
        CAST(Mark / 10.0 AS DECIMAL(5, 2)) AS Percentage
    FROM dbo.FinalMarks
),
Categorized AS (
    SELECT
        Percentage,
        CASE
            WHEN Percentage > 94.00 THEN 'Summa Cum Laude'
            WHEN Percentage >= 85.00 AND Percentage < 94.00 THEN 'Magna Cum Laude'
            WHEN Percentage >= 75.00 AND Percentage < 84.00 THEN 'Cum Laude'
        END AS Distinction,
        CASE
            WHEN Percentage > 94.00 THEN 1
            WHEN Percentage >= 85.00 AND Percentage < 94.00 THEN 2
            WHEN Percentage >= 75.00 AND Percentage < 84.00 THEN 3
        END AS SortOrder
    FROM PercentageMarks
    WHERE Percentage > 94.00
       OR (Percentage >= 85.00 AND Percentage < 94.00)
       OR (Percentage >= 75.00 AND Percentage < 84.00)
)
SELECT
    Distinction,
    COUNT(*) AS [Count],
    CAST(MAX(Percentage) AS DECIMAL(10, 2)) AS [Highest Percentage in range]
FROM Categorized
GROUP BY Distinction, SortOrder
ORDER BY SortOrder;
GO

    CASE
        WHEN CAST(SUBSTRING(IDNumber, 7, 4) AS INT) >= 5000 THEN 'Mr'
        ELSE 'Ms'
    END AS Title,
    [Name] + ' ' + Surname AS [Student Full Name],
    Mark
FROM dbo.FinalMarks
